# Стартовый скрипт для этапа 3: проверка всех реализованных команд.
# Комментарии начинаются с символа # (синтаксис Python).

# --- Работа с VFS (служебная команда) ---
vfs-info                    # имя VFS, число каталогов/файлов и дерево

# --- Команды-заглушки с разными аргументами ---
ls                          # без аргументов
ls -l "My Documents"        # аргумент в двойных кавычках
cd 'home/user'              # аргумент в одинарных кавычках
cd "#not a comment"         # символ # внутри кавычек — не комментарий

# --- Обработка ошибок ---
foo bar                     # неизвестная команда
vfs-info extra              # лишние аргументы у vfs-info
exit 1                      # лишние аргументы у exit
# следующая строка содержит незакрытую кавычку
cd "unclosed
