# Стартовый скрипт для этапа 2.
# Комментарии начинаются с символа # (синтаксис Python).

ls                          # заглушка без аргументов
ls -l "My Folder"           # аргумент в двойных кавычках
cd 'docs/my files'          # аргумент в одинарных кавычках
cd "#not a comment"         # символ # внутри кавычек — не комментарий

# Обработка ошибок
foo bar                     # неизвестная команда
# следующая строка содержит незакрытую кавычку
cd "unclosed
exit 1                      # лишние аргументы у exit
